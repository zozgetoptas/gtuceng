Part 2: Triangle Validation and Perimeter

#include <stdio.h>

/* Helper Function: Checks if side lengths form a valid triangle */
int is_valid_triangle(double a, double b, double c) {
    /* Logical expression: sum of any two sides must be > third side */
    if ((a + b > c) && (a + c > b) && (b + c > a)) {
        return 1; // Valid
    } else {
        return 0; // Invalid
    }
}

/* Main Logic Function: Calls the helper and processes results */
void process_triangle(double a, double b, double c) {
    /* Nested function call: calling Function A from Function B */
    if (is_valid_triangle(a, b, c)) {
        double perimeter = a + b + c;
        printf("Triangle is valid! Perimeter: %.2f\n", perimeter);
    } else {
        /* Error message for invalid input as per Part 2 */
        printf("Error: These sides cannot form a triangle.\n");
    }
}

int main(void) {
    double side1, side2, side3;

    /* Driver logic: Reading from user */
    printf("Enter three side lengths: ");
    scanf("%lf %lf %lf", &side1, &side2, &side3);

    /* Call the main logic function */
    process_triangle(side1, side2, side3);

    return 0;
}
