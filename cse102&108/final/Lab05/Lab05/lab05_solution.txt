#include <stdio.h>
#include <math.h>

#define SIZE 10

/* Menu enumeration for main operations */
enum MenuOption {SIMPSON = 1, CONVOLUTION};

/* Function selection enumeration */
enum FuncChoice {POLYNOMIAL = 1, COSINE};

/* Kernel type enumeration */
enum KernelType {EDGE_DETECTION = 1, SMOOTHING, SHARPENING};

/* Error enumeration for integration */
enum ErrorType {SUCCESS, INVALID_BOUNDS, NOT_POSITIVE, NOT_EVEN};

/* Function prototypes */
double simpson(double f(double), double a, double b, int n, enum ErrorType *error);

/* Mathematical functions that can be passed as parameters */
double polynomial(double x);
double cosineFunc(double x);

void convolution(int arr[], int size, int kernel[3], int result[]);

int main()
{
    enum MenuOption choice;
    enum FuncChoice funcChoice;
    enum KernelType kernelChoice;
    enum ErrorType error;

    int arr[SIZE] = {1, 3, 2, 5, 4, 7, 6, 9, 8, 10};
    int result[SIZE];

    /* Define 3 different kernels locally */
    int kernel_edge[3] = {1, 0, -1};      /* Edge detection */
    int kernel_smooth[3] = {1, 2, 1};     /* Smoothing/Average */
    int kernel_sharp[3] = {-1, 2, -1};    /* Sharpening */

    double a, b;
    int n;
    double integral;

    printf("\n=============================\n");
    printf("         MAIN MENU           \n");
    printf("=============================\n");
    printf("1. Integral Calculation with Simpson's Rule\n");
    printf("2. Convolution Filter for 1D Array\n");
    printf("Enter your choice: ");

    scanf("%d", &choice);

    switch(choice)
    {
        case SIMPSON:

            /* Function selection menu using enum */
            printf("\nSelect function to integrate:\n");
            printf("1. f(x) = x^3 + 2x + 1\n");
            printf("2. f(x) = cos(x)\n");
            printf("Enter function choice: ");
            scanf("%d", &funcChoice);

            printf("\nEnter lower bound a: ");
            scanf("%lf", &a);

            printf("Enter upper bound b: ");
            scanf("%lf", &b);

            printf("Enter number of intervals n: ");
            scanf("%d", &n);

            /* Call integration directly with chosen function */
            switch(funcChoice)
            {
                case POLYNOMIAL:
                    integral = simpson(polynomial, a, b, n, &error);
                    break;
                case COSINE:
                    integral = simpson(cosineFunc, a, b, n, &error);
                    break;
                default:
                    printf("Invalid function selection! Defaulting to POLYNOMIAL function.\n");
                    integral = simpson(polynomial, a, b, n, &error);
                    break;
            }

            /* Display results with error handling */
            if(error == SUCCESS)
            {
                printf("\nApproximate integral = %.6lf\n", integral);
                 printf("Status: Success!\n");
            }
            else if(error == INVALID_BOUNDS)
            {
                printf("\nError: Lower bound must be less than upper bound (a < b)!\n");
                printf("Status: INVALID_BOUNDS\n");
            }
            else if(error == NOT_POSITIVE)
            {
                printf("\nError: n must be positive!\n");
                printf("Status: NOT_POSITIVE\n");
            }
            else if(error == NOT_EVEN)
            {
                printf("\nError: n must be even for Simpson's Rule!\n");
                printf("Status: NOT_EVEN\n");
            }

            break;

        case CONVOLUTION:

            /* Kernel selection menu using enum */
            printf("\nSelect kernel type:\n");
            printf("1. Edge Detection [1, 0, -1]\n");
            printf("2. Smoothing [1, 2, 1]\n");
            printf("3. Sharpening [-1, 2, -1]\n");
            printf("Enter kernel choice: ");
            scanf("%d", &kernelChoice);

            /* Call convolution function directly in cases with appropriate kernel arrays */
            switch(kernelChoice)
            {
                case EDGE_DETECTION:
                    convolution(arr, SIZE, kernel_edge, result);
                    printf("\nSelected Kernel: [1, 0, -1]\n");
                    break;
                case SMOOTHING:
                    printf("\nSelected Kernel: [1, 2, 1]\n");
                    convolution(arr, SIZE, kernel_smooth, result);
                    break;
                case SHARPENING:
                    printf("\nSelected Kernel: [-1, 2, -1]\n");
                    convolution(arr, SIZE, kernel_sharp, result);
                    break;
                default:
                    printf("Invalid kernel selection! Defaulting to Edge Detection.\n");
                    printf("\nSelected Kernel: [1, 0, -1]\n");
                    convolution(arr, SIZE, kernel_edge, result);
                    break;
            }


            printf("Convolution result:\n");
            int valid_result_size = SIZE - 2;
            for(int i = 0; i < valid_result_size; i++)
                printf("%d ", result[i]);
            printf("\n");

            break;

        default:
            printf("Invalid menu option\n");
    }

    return 0;
}

/* ========== Mathematical Functions (can be passed as parameters) ========== */

double polynomial(double x)
{
    return x * x * x + 2 * x + 1;
}

double cosineFunc(double x)
{
    return cos(x);
}

/* ========== Simpson's Rule Implementation ========== */
double simpson(double f(double), double a, double b, int n, enum ErrorType *error)
{
    /* Validate bounds FIRST */
    if(a >= b)
    {
        *error = INVALID_BOUNDS;
        return -1.0;
    }

    /* Validate n - must be positive */
    if(n <= 0)
    {
        *error = NOT_POSITIVE;
        return -1.0;
    }

    /* Validate n - must be even for Simpson's Rule */
    if(n % 2 != 0)
    {
        *error = NOT_EVEN;
        return -1.0;
    }

    double h = (b - a) / n;
    double sum = f(a) + f(b);
    double x;

    for(int i = 1; i < n; i++)
    {
        x = a + i * h;

        if(i % 2 == 0)
            sum += 2 * f(x);
        else
            sum += 4 * f(x);
    }

    *error = SUCCESS;

    return (h / 3.0) * sum;
}

/* ========== Convolution Filter ========== */
void convolution(int arr[], int size, int kernel[3], int result[])
{
    /* Loop from 0 to size-2 */
    for(int i = 0; i < size - 2; i++)
    {
        /* Adjusted indices: i, i+1, i+2 */
        result[i] =
            arr[i] * kernel[0] +
            arr[i + 1] * kernel[1] +
            arr[i + 2] * kernel[2];
    }
}
